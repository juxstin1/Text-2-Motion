<script>
document.fonts.ready.then(() => {
  const tl = gsap.timeline({ paused: true });

  // 1. The Stamp Slam
  // We animate the badge from its off-screen position to its target.
  // We use a fast power4.in to give it "weight" and a back.out to settle it.
  tl.fromTo("#badge", 
    { 
      y: -300, 
      rotation: 15, 
      autoAlpha: 0,
      scale: 1.2 
    },
    { 
      y: 0, 
      rotation: 0, 
      autoAlpha: 1, 
      scale: 1, 
      duration: 0.35, 
      ease: "power4.in" 
    }, 
    0
  )
  // 2. The Impact (The "Stamp" moment)
  // A tiny, high-frequency shake on the wrapper.
  .to("#content-wrapper", {
    x: 4,
    y: -2,
    duration: 0.04,
    ease: "power2.out"
  }, 0.35)
  .to("#content-wrapper", {
    x: -4,
    y: 2,
    duration: 0.04,
    ease: "power2.out"
  }, 0.39)
  .to("#content-wrapper", {
    x: 2,
    y: -1,
    duration: 0.04,
    ease: "power2.out"
  }, 0.43)
  .to("#content-wrapper", {
    x: 0,
    y: 0,
    duration: 0.04,
    ease: "power2.out"
  }, 0.47)
  
  // 3. The Settle
  // The badge does a tiny squash and stretch to "stick" to the card.
  .to("#badge", {
    scaleX: 1.1,
    scaleY: 0.9,
    duration: 0.1,
    ease: "power2.out"
  }, 0.47)
  .to("#badge", {
    scaleX: 1,
    scaleY: 1,
    duration: 0.4,
    ease: "back.out(2)"
  }, 0.57)
  
  // 4. Final hold
  .to({}, { duration: 1.5 });

  window.TL = tl;
  if (location.search.includes("play")) tl.play();
});
</script>
